## 模板教程

```html
<!DOCTYPE html>
<html>
<head>
  <meta chatset = "UTF-8"> <!-- 设置 -->
  <title>网页标题</title> <!-- 网页栏标题 -->
</head> 
<body>
  <!-- 多级标题 -->
  <h1>内容标题</h1>
  <!-- 段落 -->
  <p>段落1<br>段落2</p>
  <!-- 分割线 -->
  <hr color="red" width="300px" size="200px" align="left">
  <!--图片设置-->
  <img src="../../static/images/background.jpg" alt="备注" width="300px">
  <!-- 超文本链接 -->
  <a href="https://mife-user.github.io/">链接文本(可以是任意内容，如图片)</a>
  <!-- 超文本的图片链接 -->
  <a href="https://mife-user.github.io/">
    <img src="../../static/images/background.jpg" alt="备注" width="100px">
  </a>
  <!-- 文本标签 -->
  <em>  定义着重文字</em>
  <b> 定义粗体文字<b>
  <i> 定义斜体字<i>
  <strong>  定义加重语气</strong>
  <del> 定义删除字</del>
  <span> 元素没有特定含义<span>
  <!-- 有序列表,type表示以什么为序号 -->
  <ol type="1"> 
    <li>第一个</li>
    <li>第二个</li>
  <!-- 无序列表,type为实心圆，空心圆等(disc,circle,square,none) -->
  <ul type="">
    <li>one</li>
    <li>two</li>
  </ul>
  <!-- 表格 -->
   <table border="1" width="400px" height="200px">
    <!-- 一行 -->
    <tr> 
      <!-- 一列 -->
      <td>1</td>
      <td>2</td>
    </tr>
  </table>
  <!-- 合并单元格 -->
  <table>
    <tr>
      <td colspan="2">行合并</td>
      <td rowspan="2">水平合并</td>
    </tr>
  </table>
  <!-- Form表单 -->
  <form action="url" method="get|post" name="myform">
  <!-- 表单元素 -->
    <!-- 输入按钮-->
    用户名:<input type="text">
    密码<input type="password" name="pwd">
    <!-- 确认按钮-->
    <input type="submit" value="登录文本">
    <button>按钮</button>
  </form>
  <!-- 块元素与内联元素(即是否换行) -->
  <!-- HTML5新增标签 -->
  <!-- 容器标签 -->
  <div>主导航(容器名)</div>
</body>  
</html>
```
